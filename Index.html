    <script>

        document.addEventListener('DOMContentLoaded', () => {

            const hero = document.getElementById('hero');

            const heroLiquidOverlay = document.getElementById('hero-liquid-overlay');

            const titleElement = document.getElementById('interactive-title');

            const pageHeader = document.getElementById('page-header');

            const themeTrigger = document.getElementById('header-theme-trigger');

            const hoverNudge = document.getElementById('hover-nudge');


            pageHeader.classList.add('theme-dark');

            

            const text = titleElement.innerText;

            titleElement.innerHTML = '';

            text.split('').forEach(char => {

                const span = document.createElement('span');

                span.innerText = char;

                titleElement.appendChild(span);

            });

            const letterSpans = Array.from(titleElement.children);


            let lastMouseX = 0;

            let mouseVelocityX = 0;

            let lastTime = Date.now();


            hero.addEventListener('mousemove', (e) => {

                const rect = hero.getBoundingClientRect();

                const mouseX = e.clientX - rect.left;

                const mouseY = e.clientY - rect.top;


                heroLiquidOverlay.style.setProperty('--x', `${mouseX}px`);

                heroLiquidOverlay.style.setProperty('--y', `${mouseY}px`);


                const now = Date.now();

                const dt = now - lastTime;

                const dx = mouseX - lastMouseX;

                mouseVelocityX = dt > 0 ? dx / dt : 0;

                lastMouseX = mouseX;

                lastTime = now;


                let cursorIsNearTitle = false;

                letterSpans.forEach(span => {

                    const spanRect = span.getBoundingClientRect();

                    const spanCenterX = (spanRect.left + spanRect.right) / 2 - rect.left;

                    const spanCenterY = (spanRect.top + spanRect.bottom) / 2 - rect.top;

                    const distX = spanCenterX - mouseX;

                    const distY = spanCenterY - mouseY;

                    const distance = Math.sqrt(distX * distX + distY * distY);

                    const maxDist = 100;


                    if (distance < maxDist) {

                        cursorIsNearTitle = true; // Set flag if cursor is near any letter

                        const force = 1 - (distance / maxDist);

                        const moveX = (distX / distance) * force * 25;

                        const moveY = (distY / distance) * force * 25;

                        const rotate = mouseVelocityX * 2 * force; 

                        span.style.transform = `translate(${moveX}px, ${moveY}px) rotate(${rotate}deg)`;

                    } else {

                        span.style.transform = 'translate(0,0) rotate(0)';

                    }

                });


                // --- UPDATED: Nudge visibility logic ---

                if(cursorIsNearTitle) {

                    hoverNudge.classList.add('fade-out');

                } else {

                    hoverNudge.classList.remove('fade-out');

                }

            });


            hero.addEventListener('mouseleave', () => {

                heroLiquidOverlay.style.setProperty('--x', `-100%`);

                heroLiquidOverlay.style.setProperty('--y', `-100%`);

                letterSpans.forEach(span => {

                    span.style.transform = 'translate(0,0) rotate(0)';

                });

                // Make sure nudge is visible when mouse leaves hero

                hoverNudge.classList.remove('fade-out');

            });


            const observerOptions = {

                root: null,

                rootMargin: "-80px 0px 0px 0px",

                threshold: 0

            };


            const themeObserver = new IntersectionObserver((entries) => {

                entries.forEach(entry => {

                    if (entry.isIntersecting) {

                        pageHeader.classList.remove('theme-light');

                        pageHeader.classList.add('theme-dark');

                    } else {

                        pageHeader.classList.remove('theme-dark');

                        pageHeader.classList.add('theme-light');

                    }

                });

            }, observerOptions);


            themeObserver.observe(themeTrigger);

        });

    </script>
